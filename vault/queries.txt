**Basic SELECT and exploring the data**
```sql
-- how many rows in each table
SELECT COUNT(*) FROM `bigquery-public-data.thelook_ecommerce.users`
SELECT COUNT(*) FROM `bigquery-public-data.thelook_ecommerce.orders`

-- preview the data
SELECT * FROM `bigquery-public-data.thelook_ecommerce.users` LIMIT 10
SELECT * FROM `bigquery-public-data.thelook_ecommerce.products` LIMIT 10
```

---

**Filtering and sorting**
```sql
-- orders with a specific status
SELECT * FROM `bigquery-public-data.thelook_ecommerce.orders`
WHERE status = 'Shipped'
ORDER BY created_at DESC
LIMIT 20

-- products in a specific category
SELECT name, brand, retail_price
FROM `bigquery-public-data.thelook_ecommerce.products`
WHERE category = 'Jeans'
ORDER BY retail_price DESC
```

---

**Aggregations and grouping**
```sql
-- orders by status
SELECT status, COUNT(*) as total_orders
FROM `bigquery-public-data.thelook_ecommerce.orders`
GROUP BY status
ORDER BY total_orders DESC

-- average sale price by product category
SELECT product_category, ROUND(AVG(sale_price), 2) as avg_sale_price
FROM `bigquery-public-data.thelook_ecommerce.order_items`
GROUP BY product_category
ORDER BY avg_sale_price DESC

-- top 10 brands by number of products
SELECT brand, COUNT(*) as num_products
FROM `bigquery-public-data.thelook_ecommerce.products`
GROUP BY brand
ORDER BY num_products DESC
LIMIT 10
```

---

**Joins**
```sql
-- orders with user details
SELECT o.order_id, u.first_name, u.last_name, u.country, o.status, o.created_at
FROM `bigquery-public-data.thelook_ecommerce.orders` o
JOIN `bigquery-public-data.thelook_ecommerce.users` u ON o.user_id = u.id
LIMIT 20

-- order items with product details
SELECT oi.order_id, p.name, p.brand, p.category, oi.sale_price
FROM `bigquery-public-data.thelook_ecommerce.order_items` oi
JOIN `bigquery-public-data.thelook_ecommerce.products` p ON oi.product_id = p.id
LIMIT 20

-- multi-table join - orders, users and products
SELECT o.order_id, u.first_name, u.last_name, p.name as product, p.category, oi.sale_price
FROM `bigquery-public-data.thelook_ecommerce.orders` o
JOIN `bigquery-public-data.thelook_ecommerce.users` u ON o.user_id = u.id
JOIN `bigquery-public-data.thelook_ecommerce.order_items` oi ON o.order_id = oi.order_id
JOIN `bigquery-public-data.thelook_ecommerce.products` p ON oi.product_id = p.id
LIMIT 20
```

---

**Date functions**
```sql
-- orders by year
SELECT EXTRACT(YEAR FROM created_at) as year, COUNT(*) as total_orders
FROM `bigquery-public-data.thelook_ecommerce.orders`
GROUP BY year
ORDER BY year

-- orders by month
SELECT FORMAT_TIMESTAMP('%Y-%m', created_at) as month, COUNT(*) as total_orders
FROM `bigquery-public-data.thelook_ecommerce.orders`
GROUP BY month
ORDER BY month
```
